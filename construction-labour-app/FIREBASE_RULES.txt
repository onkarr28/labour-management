rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    // Labours data (attendance, payments, etc.)
    match /labours/{labourId} {
      allow read, write: if true;
    }

    // Contractors master data
    match /contractors/{contractorId} {
      allow read, write: if true;
    }

    // Worker login credentials and mapping to contractors
    match /workers/{workerId} {
      allow read, write: if true;
    }

    // Fallback: allow all other collections
    match /{document=**} {
      allow read, write: if true;
    }
  }
}
