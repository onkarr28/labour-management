rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // Labours Collection - Full read/write access for all contractors
    match /labours/{labourId} {
      allow read, write: if true;
      
      // Allow access to subcollections
      match /{document=**} {
        allow read, write: if true;
      }
    }
    
    // Contractors Collection - Full read/write access
    match /contractors/{contractorId} {
      allow read, write: if true;
    }
    
    // Worker Credentials - Full read/write access for authentication
    match /worker_credentials/{workerId} {
      allow read, write: if true;
    }
    
    // Any other collection - Full access
    match /{document=**} {
      allow read, write: if true;
    }
  }
}
